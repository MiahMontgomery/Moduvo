// services/evidence/collector.ts
import fs from "fs";
import path from "path";
import { artifactsRoot } from "../../server/bootstrap/artifacts";

export async function saveJsonArtifact(runId: string, name: string, obj: any) {
  const dir = path.join(artifactsRoot, runId);
  fs.mkdirSync(dir, { recursive: true });
  const file = path.join(dir, `${Date.now()}_${name}.json`);
  fs.writeFileSync(file, JSON.stringify(obj, null, 2), "utf-8");
  return `/artifacts/${runId}/${path.basename(file)}`;
}

export async function captureScreenshot(runId: string, page: any, name = "shot") {
  const dir = path.join(artifactsRoot, runId);
  fs.mkdirSync(dir, { recursive: true });
  const file = `${Date.now()}_${name}.png`;
  const abs = path.join(dir, file);
  await page.screenshot({ path: abs, fullPage: true });
  return `/artifacts/${runId}/${file}`;
}

// services/trest/guard.ts
import { db } from "../../server/db";

export async function withBudgetGuard<T>(projectId: string, estCostCents: number, fn: () => Promise<T>) {
  const { spent = 0, cap = 0 } = await db.get(`
    SELECT COALESCE(SUM(amount_cents),0) as spent, (SELECT monthly_cap_cents FROM budget_guards WHERE project_id = ?) as cap
  `,[projectId]) ?? {};
  if (cap && spent + estCostCents > cap) {
    await db.run("INSERT INTO budget_events(project_id, kind, detail) VALUES(?,?,?)",
      [projectId, "BUDGET_EXCEEDED", JSON.stringify({ spent, cap, estCostCents })]);
    throw new Error("BUDGET_EXCEEDED");
  }
  const out = await fn();
  // record actual cost here if you meter per-call
  return out;
}

// services/email/gmail.ts
import { google } from "googleapis";

export async function createDraft({ accessToken, to, subject, html }:{
  accessToken: string; to: string; subject: string; html: string;
}) {
  const auth = new google.auth.OAuth2();
  auth.setCredentials({ access_token: accessToken });
  const gmail = google.gmail({ version: "v1", auth });
  const message =
    `To: ${to}\r\n` +
    `Subject: ${subject}\r\n` +
    `Content-Type: text/html; charset=UTF-8\r\n\r\n` +
    html;
  const encoded = Buffer.from(message).toString("base64").replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
  const { data } = await gmail.users.drafts.create({
    userId: "me",
    requestBody: { message: { raw: encoded } }
  });
  return data; // includes id, message.threadId
}
