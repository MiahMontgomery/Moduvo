import nodemailer from "nodemailer";
let transporter: nodemailer.Transporter | null = null;
export async function createEmailTransporter() {
  if (transporter) return transporter;
  transporter = nodemailer.createTransport({
    host: process.env.SMTP_HOST!,
    port: Number(process.env.SMTP_PORT || 587),
    secure: false,
    auth: { user: process.env.SMTP_USER!, pass: process.env.SMTP_PASS! }
  });
  try { await transporter.verify(); } catch { /* log-only */ }
  return transporter;
}
export async function sendQuoteEmail(to: string, subject: string, html: string) {
  const t = await createEmailTransporter();
  return t.sendMail({ from: process.env.FROM_EMAIL!, to, subject, html });
}

#!/usr/bin/env bash
set -euo pipefail

echo "== Grep =="
grep -RIn "helmet" server/index.ts || true
grep -RIn "cors(" server/index.ts || true
grep -RIn "compression" server/index.ts || true
grep -RIn "rateLimit" server/index.ts || true
grep -RIn "requireAdmin" server || true
grep -RIn "/api/admin" server/routes.* || true
grep -RIn "calculateQuote" server/routes.* server/pricing.* || true
grep -RIn "smtp.migadu.com" -n server || true

echo "== Typecheck & build =="
npm run typecheck || true
npm run build

echo "== Start app =="
PORT=8080 NODE_ENV=production LOG_LEVEL=info \
CORS_ORIGIN=https://moduvo.to,https://www.moduvo.to \
JWT_SECRET=devdevdevdevdevdevdevdevdevdevdevdevdevdevdevdevdevdevdevdev \
SESSION_SECRET=devdevdevdevdevdevdevdevdevdevdevdevdevdevdevdevdevdevdev \
DATABASE_URL=${DATABASE_URL:-postgresql://moduvo:password@127.0.0.1:5432/moduvo} \
SMTP_HOST=smtp.migadu.com SMTP_PORT=587 SMTP_USER=quotes@moduvo.to SMTP_PASS=xxx \
FROM_EMAIL="Moduvo Quotes <quotes@moduvo.to>" \
GCS_BUCKET=${GCS_BUCKET:-} GOOGLE_APPLICATION_CREDENTIALS=${GOOGLE_APPLICATION_CREDENTIALS:-/etc/moduvo/gcp.json} \
node server/dist/index.js > /tmp/moduvo.out 2>&1 &

APP_PID=$!

echo "== Wait for healthz =="
for i in {1..25}; do
  if curl -sf http://127.0.0.1:8080/healthz >/dev/null; then echo healthy; break; fi
  sleep 1
done
curl -sf http://127.0.0.1:8080/healthz || { echo 'Health check failed'; tail -n 150 /tmp/moduvo.out; exit 1; }

echo "== Headers (health) =="
curl -sI http://127.0.0.1:8080/healthz | sed -n '1,30p'

echo "== RBAC negative =="
curl -s -o /dev/null -w "%{http_code}\n" http://127.0.0.1:8080/api/admin/quotes

echo "== SPA =="
test -d client/dist && echo DIST_OK && ls -la client/dist | sed -n '1,80p'

echo "== Pricing fingerprint =="
sed -n '1,120p' server/pricing.ts | sed -n '1,40p'

echo "== Stop =="
kill $APP_PID || true
wait $APP_PID 2>/dev/null || true
echo OK
