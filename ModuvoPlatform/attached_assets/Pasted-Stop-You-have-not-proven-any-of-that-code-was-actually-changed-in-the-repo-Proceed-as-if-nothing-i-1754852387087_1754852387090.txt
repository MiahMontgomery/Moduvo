# 6) Admin RBAC guard

Create a reusable guard and apply to all `/api/admin/*`.

```ts
// server/middleware/auth.ts
export function requireAuth(req,res,next){
  if(!req.user) return res.status(401).json({error:"unauthenticated"});
  next();
}
export function requireAdmin(req,res,next){
  if(!req.user || !["admin","staff"].includes(req.user.role)) {
    return res.status(403).json({error:"forbidden"});
  }
  next();
}

// server/routes.ts (example)
router.get("/api/admin/quotes", requireAuth, requireAdmin, async (req,res)=>{ /*...*/ });
```

# 7) Migadu email correctness

Use STARTTLS 587. Do not hardcode from address.

```ts
// server/email.ts
import nodemailer from "nodemailer";
export const transporter = nodemailer.createTransport({
  host: process.env.SMTP_HOST!,
  port: Number(process.env.SMTP_PORT || 587),
  secure: false,
  auth: { user: process.env.SMTP_USER!, pass: process.env.SMTP_PASS! },

await transporter.verify(); // run at boot, log result
```

# 8) Nginx reverse proxy

`/etc/nginx/sites-available/moduvo.conf`

```nginx
server {
  listen 80;
  server_name moduvo.to www.moduvo.to;
  location /.well-known/acme-challenge/ { root /var/www/html; }
  location / { return 301 https://moduvo.to$request_uri; }
}

server {
  listen 443 ssl http2;
  server_name moduvo.to www.moduvo.to;

  ssl_certificate /etc/letsencrypt/live/moduvo.to/fullchain.pem;
  ssl_certificate_key /etc/letsencrypt/live/moduvo.to/privkey.pem;

  # Security headers
  add_header X-Frame-Options DENY;
  add_header X-Content-Type-Options nosniff;
  add_header Referrer-Policy strict-origin-when-cross-origin;
  add_header Permissions-Policy "camera=(), microphone=(), geolocation=()";

  client_max_body_size 30m;

  location / {
    proxy_pass http://127.0.0.1:8080;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_read_timeout 120;
  }

  location /healthz {
    proxy_pass http://127.0.0.1:8080/healthz;
  }

  # serve static client if API also serves SPA
  location /assets/ { alias /opt/moduvo/client/dist/assets/; }
}
```
